
# ==========================================================
# DHAN CRUDEOILM - ORDER MANAGER
# VERSION 4.0 - STATE STORE V5 COMPATIBLE
#
# ONE LOT PER ENTRY
# UNIQUE CORRELATION IDS
# WRITE-AHEAD ORDER INTENTS
# UNCERTAIN SUBMISSION PROTECTION
# MANUAL CANCELLATION LOCK AWARE
# BUY / SELL GRID SUPPORT
# COMMON SL ORDER SUPPORT
#
# NO AUTOMATIC RETRY
# NO CANCEL AND REPLACE
# NO DUPLICATE SUBMISSION
# ==========================================================

import hashlib
from datetime import datetime

import config


class OrderManager:

    ACTIVE_STATUSES = {
        "PENDING",
        "TRANSIT",
        "PART_TRADED",
        "OPEN",
        "TRADED",
    }

    UNRESOLVED_STATUSES = {
        "SUBMISSION_PENDING",
        "SUBMISSION_UNCERTAIN",
        "VERIFICATION_REQUIRED",
    }

    def __init__(
        self,
        client,
        state_store,
        state,
    ):

        self.client = client
        self.state_store = state_store
        self.state = state

        if not isinstance(self.state, dict):
            raise RuntimeError(
                "OrderManager requires a valid state dictionary."
            )

        self.state.setdefault("orders", {})

        orders = self.state["orders"]

        orders.setdefault("pending_entry_orders", {})
        orders.setdefault("stop_loss_orders", {})
        orders.setdefault("order_fill_totals", {})
        orders.setdefault("manual_cancelled_orders", {})

    # ======================================================
    # HELPERS
    # ======================================================

    def _now(self):

        return datetime.now().isoformat()

    def _positive_integer(self, value, name):

        if isinstance(value, bool):
            raise RuntimeError(
                "Invalid " + name
            )

        try:
            result = int(value)
        except (TypeError, ValueError):
            raise RuntimeError(
                "Invalid " + name
            )

        if result <= 0 or result != value:
            raise RuntimeError(
                "Invalid " + name
            )

        return result

    def _positive_price(self, value, name):

        try:
            result = float(value)
        except (TypeError, ValueError):
            raise RuntimeError(
                "Invalid " + name
            )

        if result <= 0:
            raise RuntimeError(
                "Invalid " + name
            )

        return result

    def _save(self):

        self.state_store.save(self.state)

    # ======================================================
    # CYCLE VALIDATION
    # ======================================================

    def _validate_cycle(self):

        strategy = self.state.get(
            "strategy",
            {},
        )

        if not isinstance(strategy, dict):
            raise RuntimeError(
                "Strategy state is invalid."
            )

        if strategy.get("cycle_active") is not True:
            raise RuntimeError(
                "No active execution cycle."
            )

        if strategy.get("cycle_closed") is not False:
            raise RuntimeError(
                "Execution cycle is closed."
            )

        execution_id = strategy.get("execution_id")

        if (
            not isinstance(execution_id, str)
            or not execution_id.strip()
        ):
            raise RuntimeError(
                "Execution ID is missing."
            )

        active = self.state.get(
            "active_contract",
            {},
        )

        if str(active.get("symbol")) != str(
            config.ACTIVE_SYMBOL
        ):
            raise RuntimeError(
                "Active symbol mismatch."
            )

        if str(active.get("security_id")) != str(
            config.ACTIVE_SECURITY_ID
        ):
            raise RuntimeError(
                "Active security ID mismatch."
            )

        if str(active.get("expiry")) != str(
            config.ACTIVE_EXPIRY
        ):
            raise RuntimeError(
                "Active expiry mismatch."
            )

        if str(active.get("exchange_segment")) != str(
            config.EXCHANGE_SEGMENT
        ):
            raise RuntimeError(
                "Active exchange segment mismatch."
            )

        return execution_id

    # ======================================================
    # CORRELATION ID
    # ======================================================

    def _correlation_id(self, role, sequence):

        execution_id = self._validate_cycle()

        raw = (
            str(execution_id)
            + "|"
            + str(role)
            + "|"
            + str(sequence)
        )

        digest = hashlib.sha256(
            raw.encode("utf-8")
        ).hexdigest()[:20].upper()

        prefix = (
            "SSENTRY"
            if role == "ENTRY"
            else "SSSTOP"
        )

        return prefix + digest

    # ======================================================
    # EXISTING INTENT CHECK
    # ======================================================

    def _find_existing_correlation(
        self,
        group,
        correlation_id,
    ):

        for key, record in group.items():

            if not isinstance(record, dict):
                continue

            saved_id = (
                record.get("correlationId")
                or record.get("correlation_id")
                or record.get("payload", {}).get("correlationId")
            )

            if str(saved_id) == str(correlation_id):
                return key, record

        return None, None

    def _has_unresolved_intent(self, group):

        for record in group.values():

            if not isinstance(record, dict):
                raise RuntimeError(
                    "Invalid saved order intent."
                )

            status = str(
                record.get("status", "")
            ).upper()

            if status in self.UNRESOLVED_STATUSES:
                return True

        return False

    # ======================================================
    # MANUAL CANCELLATION LOCK
    # ======================================================

    def _check_cancellation_lock(self, role):

        lock_role = (
            "ENTRY_LIMIT"
            if role == "ENTRY"
            else "STOP_LOSS"
        )

        if self.state_store.is_locked(
            self.state,
            lock_role,
        ):
            raise RuntimeError(
                "Order blocked by manual cancellation lock: "
                + lock_role
            )

    # ======================================================
    # ORDER INTENT SAVE
    # ======================================================

    def _save_intent(
        self,
        role,
        sequence,
        correlation_id,
        payload,
    ):

        orders = self.state["orders"]

        group_name = (
            "pending_entry_orders"
            if role == "ENTRY"
            else "stop_loss_orders"
        )

        group = orders[group_name]

        existing_key, existing = self._find_existing_correlation(
            group,
            correlation_id,
        )

        if existing is not None:

            raise RuntimeError(
                "Duplicate or previously submitted order intent exists. "
                "Automatic resubmission is prohibited. "
                "Correlation ID: "
                + str(correlation_id)
            )

        if self._has_unresolved_intent(group):
            raise RuntimeError(
                "Another unresolved order intent exists in "
                + group_name
            )

        active = self.state["active_contract"]

        record = {
            "correlationId": correlation_id,
            "correlation_id": correlation_id,
            "order_id": None,
            "role": role,
            "sequence": int(sequence),
            "status": "SUBMISSION_PENDING",
            "quantity": int(payload["quantity"]),
            "product_type": str(payload["productType"]).upper(),
            "transaction_type": str(payload["transactionType"]).upper(),
            "security_id": str(active["security_id"]),
            "exchange_segment": str(active["exchange_segment"]),
            "payload": dict(payload),
            "filled_qty": 0,
            "average_price": 0.0,
            "created_at": self._now(),
            "updated_at": self._now(),
        }

        group[correlation_id] = record

        # Save before sending any broker request.
        self._save()

        return record

    # ======================================================
    # SUBMISSION RESPONSE
    # ======================================================

    def _extract_order_id(self, response):

        if not isinstance(response, dict):
            return None

        data = response.get("data")
        if not isinstance(data, dict):
            data = {}

        order_id = (
            response.get("orderId")
            or response.get("order_id")
            or data.get("orderId")
            or data.get("order_id")
        )

        if order_id is None:
            return None

        return str(order_id)

    def _mark_uncertain(
        self,
        record,
        error,
    ):

        record["status"] = "SUBMISSION_UNCERTAIN"
        record["submission_error"] = str(error)
        record["updated_at"] = self._now()

        self._save()

    # ======================================================
    # SUBMIT
    # ======================================================

    def _submit(
        self,
        role,
        sequence,
        payload,
    ):

        self._validate_cycle()

        self._check_cancellation_lock(role)

        if config.LIVE_TRADING_ENABLED is not True:
            raise RuntimeError(
                "LIVE_TRADING_ENABLED is OFF. "
                "Broker order submission blocked."
            )

        if config.ORDER_PLACEMENT_ENABLED is not True:
            raise RuntimeError(
                "ORDER_PLACEMENT_ENABLED is OFF. "
                "Broker order submission blocked."
            )

        correlation_id = self._correlation_id(
            role,
            sequence,
        )

        payload = dict(payload)

        payload["correlationId"] = correlation_id
        payload["dhanClientId"] = str(
            config.DHAN_CLIENT_ID
        )

        record = self._save_intent(
            role,
            sequence,
            correlation_id,
            payload,
        )

        # From this point onward, any uncertain outcome
        # is recorded and never automatically retried.
        try:

            response = self.client.place_order(
                payload
            )

        except Exception as error:

            self._mark_uncertain(
                record,
                error,
            )

            raise RuntimeError(
                "Order submission outcome is uncertain. "
                "No automatic retry will occur. "
                "Check broker order book manually. "
                + str(error)
            )

        if not isinstance(response, dict):

            self._mark_uncertain(
                record,
                "Invalid broker response.",
            )

            raise RuntimeError(
                "Invalid broker response. "
                "Submission outcome is uncertain."
            )

        order_status = str(
            response.get("orderStatus")
            or response.get("status")
            or ""
        ).upper()

        order_id = self._extract_order_id(
            response
        )

        if not order_id or order_status not in (
            "TRANSIT", "PENDING", "PART_TRADED",
            "TRADED", "REJECTED", "CANCELLED", "EXPIRED"
        ):
            self._mark_uncertain(
                record,
                response,
            )

            raise RuntimeError(
                "Broker response did not confirm a valid order ID/status. "
                "Intent locked against automatic retry."
            )

        record["order_id"] = order_id
        record["status"] = order_status
        record["broker_response"] = {
            "orderId": order_id,
            "orderStatus": order_status,
        }
        record["updated_at"] = self._now()

        self._save()

        return {
            "success": True,
            "order_id": order_id,
            "correlation_id": correlation_id,
            "status": order_status,
        }

    # ======================================================
    # FIRST CYCLE PRICE CALCULATION
    # ======================================================

    def calculate_first_cycle(self, ltp):

        ltp = self._positive_price(
            ltp,
            "LTP",
        )

        strategy = self.state["strategy"]

        direction = str(
            strategy.get("direction", "")
        ).upper()

        grid = self._positive_integer(
            strategy.get("grid_points"),
            "grid points",
        )

        sl_distance = self._positive_integer(
            strategy.get("stop_loss_points"),
            "stop loss points",
        )

        if direction == "SELL":

            entry = (
                int((ltp + grid - 0.0000001) // grid)
                * grid
            )

            stop_loss = entry + sl_distance
            next_grid = entry - grid

        elif direction == "BUY":

            entry = (
                int(ltp // grid)
                * grid
            )

            if entry <= 0:
                raise RuntimeError(
                    "Calculated BUY entry price is invalid."
                )

            stop_loss = entry - sl_distance
            next_grid = entry + grid

            if stop_loss <= 0:
                raise RuntimeError(
                    "Calculated BUY stop loss is invalid."
                )

        else:

            raise RuntimeError(
                "Direction must be BUY or SELL."
            )

        return {
            "entry_price": float(entry),
            "stop_loss": float(stop_loss),
            "next_grid": float(next_grid),
        }

    # ======================================================
    # ENTRY ORDER
    # ======================================================

    def submit_entry(
        self,
        sequence,
        price,
    ):

        sequence = self._positive_integer(
            sequence,
            "entry sequence",
        )

        price = self._positive_price(
            price,
            "entry price",
        )

        strategy = self.state["strategy"]

        direction = str(
            strategy.get("direction", "")
        ).upper()

        if direction not in ("BUY", "SELL"):
            raise RuntimeError(
                "Invalid entry direction."
            )

        maximum_entries = self._positive_integer(
            strategy.get("maximum_entries"),
            "maximum entries",
        )

        if sequence > maximum_entries:
            raise RuntimeError(
                "Maximum entry count reached."
            )

        quantity = self._positive_integer(
            strategy.get("quantity_per_entry"),
            "quantity per entry",
        )

        lot_size = self._positive_integer(
            strategy.get("lot_size"),
            "lot size",
        )

        lots = self._positive_integer(
            strategy.get("lots_per_entry"),
            "lots per entry",
        )

        if quantity != lot_size * lots:
            raise RuntimeError(
                "Entry quantity does not match lot configuration."
            )

        product_type = str(
            getattr(config, "PRODUCT_TYPE", "INTRADAY")
        ).upper()

        if product_type not in ("MARGIN", "INTRADAY"):
            raise RuntimeError("Configured product type must be MARGIN or INTRADAY.")

        payload = {
            "transactionType": direction,
            "exchangeSegment": str(config.EXCHANGE_SEGMENT),
            "productType": product_type,
            "orderType": "LIMIT",
            "validity": "DAY",
            "securityId": str(config.ACTIVE_SECURITY_ID),
            "quantity": quantity,
            "price": price,
            "triggerPrice": 0,
        }

        return self._submit(
            role="ENTRY",
            sequence=sequence,
            payload=payload,
        )

    # ======================================================
    # COMMON STOP LOSS ORDER
    # ======================================================

    def submit_stop_loss(
        self,
        sequence,
        quantity,
        trigger_price,
        product_type="INTRADAY",
    ):

        sequence = self._positive_integer(
            sequence,
            "stop loss sequence",
        )

        quantity = self._positive_integer(
            quantity,
            "stop loss quantity",
        )

        trigger_price = self._positive_price(
            trigger_price,
            "stop loss trigger",
        )

        strategy = self.state["strategy"]

        direction = str(
            strategy.get("direction", "")
        ).upper()

        if direction == "SELL":
            exit_direction = "BUY"

        elif direction == "BUY":
            exit_direction = "SELL"

        else:
            raise RuntimeError(
                "Invalid strategy direction."
            )

        product_type = str(product_type).upper()
        if product_type not in ("MARGIN", "INTRADAY"):
            raise RuntimeError("SL product type must be MARGIN or INTRADAY.")

        positions = self.state.get("positions", {})
        quantity_key = (
            "margin_quantity"
            if product_type == "MARGIN"
            else "intraday_quantity"
        )
        registered_quantity = self._positive_integer(
            positions.get(quantity_key, 0),
            quantity_key,
        ) if positions.get(quantity_key, 0) else 0

        if quantity != registered_quantity:
            raise RuntimeError(
                "SL quantity must equal registered open quantity for "
                + product_type + "."
            )

        payload = {
            "transactionType": exit_direction,
            "exchangeSegment": str(config.EXCHANGE_SEGMENT),
            "productType": product_type,
            "orderType": "STOP_LOSS_MARKET",
            "validity": "DAY",
            "securityId": str(config.ACTIVE_SECURITY_ID),
            "quantity": quantity,
            "price": 0,
            "triggerPrice": trigger_price,
        }

        return self._submit(
            role="STOP_LOSS",
            sequence=sequence,
            payload=payload,
        )

    def _integer_position_total(self):

        positions = self.state.get(
            "positions",
            {},
        )

        try:
            return int(
                positions.get("total_quantity", 0)
            )
        except (TypeError, ValueError):
            raise RuntimeError(
                "Registered position total is invalid."
            )
